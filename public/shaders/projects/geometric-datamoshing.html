How the shader is built

Each stage plays a role in the final image. The order matters as coordinate transformations happen before texture sampling, while masking and compositing operate on the sampled colors.
01 — UV COORDINATES

Convert the fragment position into normalized 0–1 coordinates and establish a centered coordinate system for matrix operations.

02 — MATRIX TRANSFORMS

Apply scale, rotation, shear and translation. These operations alter the image’s spatial orientation and placement.

03 — DISPLACEMENT

Sample channel values from grayscale map to offset the transformed UV coordinates. Add the horizontal sine-wave distortion.

04 — SECONDARY IMAGE

Sample the color layer on a pixel grid, offset rows, separate red and blue channels, and limit the result into fewer color levels through quanitzation.

05 — BINARY MASK

Calculate luminance from the base image and apply a threshold for it. The resulting 0/1 mask controls where the secondary image becomes more visible.

06 — COMPOSITE

Blend the secondary layer over the transformed source using an alpha value and the selected Difference or Division operation.

/*
    GEOMETRIC DATAMOSHING
    Spatial displacement / Matrix transformations
    Alpha compositing / Difference / Division
    Pixelation / Color quantization / Binary masking

    iChannel0: Original geometric composition
    iChannel1: Colorful glitch / pixel texture
    iChannel2: Grayscale displacement map
*/

uniform float uBlendMode;

//---------------- HELPER FUNCTIONS ----------------//

// Luminance for binary masking
float getBrightness(vec3 color)
{
    return dot(color, vec3(0.299, 0.587, 0.114));
}

// Random value for horizontal glitch displacement
float hash(float n)
{
    return fract(sin(n * 127.1) * 43758.5453);
}

// Color quantization
vec3 quantizeColor(vec3 color, float levels)
{
    return floor(color * levels) / levels;
}


//---------------- ALPHA COMPOSITING ----------------//

// Source-over compositing with a selected blend operation
vec4 compositeOver(vec4 base, vec3 source, float alpha, float mode)
{
    vec3 blended;

    // Difference blending
    if (mode < 0.5)
    {
        blended = abs(base.rgb - source);
    }

    // Division blending
    else
    {
        blended = base.rgb / max(source, vec3(0.08));
        blended = clamp(blended, 0.0, 1.0);
    }

    // Alpha Over operator
    float outAlpha = alpha + base.a * (1.0 - alpha);

    vec3 outColor =
        (blended * alpha +
         base.rgb * base.a * (1.0 - alpha))
        / max(outAlpha, 0.0001);

    return vec4(outColor, outAlpha);
}


//---------------- MAIN SHADER ----------------//

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    // Normalized pixel coordinates (0 to 1)
    vec2 uv = fragCoord / iResolution.xy;

    vec2 mouse = iMouse.xy / iResolution.xy;

    // Mouse interaction:
    // Click and hold to manipulate the transformation.
    float mouseActive = step(1.0, iMouse.z);

    float shearAmount = mix(
        0.18,
        (mouse.x - 0.5) * 0.8,
        mouseActive
    );

    float warpAmount = mix(
        0.025,
        mouse.y * 0.08,
        mouseActive
    );


    //---------------- MATRIX TRANSFORMATIONS ----------------//

    // Center coordinates around origin
    vec2 p = uv - 0.5;

    // Scaling matrix
    float scaleAmount = 1.0 + 0.08 * sin(iTime * 0.5);

    mat2 scaleMatrix = mat2(
        scaleAmount, 0.0,
        0.0, 1.0 / scaleAmount
    );

    // Rotation matrix
    float angle = 0.06 * sin(iTime * 0.4);

    mat2 rotationMatrix = mat2(
        cos(angle), sin(angle),
        -sin(angle), cos(angle)
    );

    // Shear matrix
    mat2 shearMatrix = mat2(
        1.0, 0.0,
        shearAmount, 1.0
    );

    // Combine matrix transformations
    mat2 transformMatrix =
        rotationMatrix * shearMatrix * scaleMatrix;

    p = transformMatrix * p;

    // Translation
    p += vec2(
        0.025 * sin(iTime * 0.6),
        0.015 * cos(iTime * 0.4)
    );

    // Return to UV space
    vec2 transformedUV = p + 0.5;


    //---------------- NONLINEAR WARPING ----------------//

    // Sample displacement texture
    vec2 displacement = texture(
        iChannel2,
        clamp(transformedUV, 0.0, 1.0)
    ).rg - 0.5;

    // Horizontal sinusoidal displacement
    float distortion =
        sin(transformedUV.y * 35.0 + iTime * 1.5)
        * warpAmount;

    // Apply texture displacement and wave distortion
    transformedUV += displacement * warpAmount * 2.0;

    transformedUV.x += distortion;

    // Clamp transformed coordinates
    transformedUV = clamp(transformedUV, 0.001, 0.999);


    //---------------- BASE IMAGE ----------------//

    // Original geometric composition after transformations
    vec4 baseColor = texture(iChannel0, transformedUV);


    //---------------- PIXELATION ----------------//

    // Pixel size controls the sampling resolution
    float pixelSize = mix(
        8.0,
        28.0,
        mouseActive * mouse.y
    );

    // Quantize fragment coordinates into pixel blocks
    vec2 pixelFrag =
        floor(fragCoord / pixelSize) * pixelSize
        + pixelSize * 0.5;

    vec2 pixelUV = pixelFrag / iResolution.xy;


    //---------------- PIXEL BLOCK DISPLACEMENT ----------------//

    // Create horizontal glitch offsets per scanline
    float row = floor(fragCoord.y / (pixelSize * 2.0));

    float rowOffset =
        (hash(row) - 0.5) * 0.18;

    // Shift individual horizontal blocks
    vec2 glitchUV = pixelUV;

    glitchUV.x += rowOffset;

    // Add a secondary horizontal wave
    glitchUV.x += sin(
        pixelUV.y * 50.0 + iTime * 2.0
    ) * 0.012;

    glitchUV = clamp(glitchUV, 0.001, 0.999);


    //---------------- COLOR TEXTURE ----------------//

    // Sample the colorful glitch image
    vec4 glitchColor = texture(iChannel1, glitchUV);

    // RGB channel separation
    float channelOffset = 0.008;

    glitchColor.r = texture(
        iChannel1,
        clamp(glitchUV + vec2(channelOffset, 0.0), 0.0, 1.0)
    ).r;

    glitchColor.b = texture(
        iChannel1,
        clamp(glitchUV - vec2(channelOffset, 0.0), 0.0, 1.0)
    ).b;


    //---------------- COLOR QUANTIZATION ----------------//

    float levels = 8.0;

    glitchColor.rgb = quantizeColor(
        glitchColor.rgb,
        levels
    );


    //---------------- BINARY TRANSFORMATIONS ----------------//

    // Threshold for brightness of original image
    float brightness = getBrightness(baseColor.rgb);

    // Binary mask: pixels become either 0 or 1
    float mask = step(0.35, brightness);

    // Use the mask to control where corruption occurs
    // A small amount of glitch remains visible outside shapes
    float glitchMask = mix(0.20, 1.0, mask);


    //---------------- ALPHA BLENDING ----------------//

    // Control transparency of the glitch layer
    float alpha = 0.65 * glitchMask;

    // The preview control selects Difference (0) or Division (1).
    float blendMode = uBlendMode;

    // Composite the corrupted image over the original
    vec4 compositeColor = compositeOver(
        baseColor,
        glitchColor.rgb,
        alpha,
        blendMode
    );


    //---------------- FINAL OUTPUT ----------------//

    fragColor = vec4(compositeColor.rgb, 1.0);
}
